class MainActivity : ComponentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge()
        setContent {
            var currentTheme by remember { mutableStateOf(ThemePreference.getCurrentTheme(this)) }
            var visualStyle by remember { mutableStateOf(ThemePreference.getVisualStyle(this)) }
            var fontScale by remember { mutableStateOf(ThemePreference.getFontScale(this)) }
            // 主题 / 风格偏好变更即时生效（SharedPreferences 监听，无需重启应用）
            DisposableEffect(Unit) {
                val listener =
                    android.content.SharedPreferences.OnSharedPreferenceChangeListener { _, key ->
                        when (key) {
                            ThemePreference.KEY_THEME_ID ->
                                currentTheme = ThemePreference.getCurrentTheme(this@MainActivity)
                            ThemePreference.KEY_VISUAL_STYLE ->
                                visualStyle = ThemePreference.getVisualStyle(this@MainActivity)
                            ThemePreference.KEY_FONT_SCALE ->
                                fontScale = ThemePreference.getFontScale(this@MainActivity)
                        }
                    }
                ThemePreference.registerListener(this@MainActivity, listener)
                onDispose { ThemePreference.unregisterListener(this@MainActivity, listener) }
            }
            PyNeonTheme(appTheme = currentTheme, visualStyle = visualStyle, fontScale = fontScale) {
                AppRoot()
            }
        }
    }
}

 * C 运行时，用 Kotlin 写的教学子集解释器。
 *
 * Android 离线装 clang 加头文件和链接器要两三百 MB，还链不上 bionic，
 * 所以这里直接解释执行。gcc 能算的跟着算，不支持的明确报错。
 *
 * Java、C++、Go、Rust 在 PLANNED 里显示暂不支持，C 是真的能跑：
 * 写 printf 有输出，能判题，能改错。
 */
object CBridge : LanguageRuntime {

    override val id: String = LanguageIds.C
    override val displayName: String = LanguageIds.displayNameOf(LanguageIds.C)
    override val fileExtension: String = "c"
    override val grammarScope: String = "source.c"

    // 标清楚是子集解释器，别让人以为是 gcc。
    override val version: String get() = "教学子集（解释执行）"

    override fun isAvailable(): Boolean = true

    override fun ensureStarted(context: Context) = Unit

    override fun runCode(
        code: String,
        stdinLines: List<String>,
        timeoutSec: Double
    ): RunResult {
        val started = System.nanoTime()
        val r = CInterpreter.run(code, stdinLines)
        return toRunResult(r, started)
    }

    override fun checkExercise(
        code: String,
        tests: List<String>,
        stdinLines: List<String>
    ): RunResult {
        val started = System.nanoTime()
        val r = CInterpreter.check(code, tests)
        val base = toRunResult(r, started)
        return base.copy(passed = r.ok)
    }

    override fun warmup(): Boolean = true

    // 写个 printf 还要求有 main。现在用有状态的 Session：
    // 函数和全局变量都留着，花括号没闭合就等下一行。

    @Volatile
    private var session: CInterpreter.Session? = null

    private fun sessionLocked(): CInterpreter.Session =
        session ?: CInterpreter.Session().also { session = it }

    override fun replStart() {
        session = CInterpreter.Session()
    }

    override fun replPush(line: String): Pair<Boolean, String> =
        synchronized(this) { sessionLocked().push(line) }

    private fun toRunResult(r: CInterpreter.Result, startedAt: Long): RunResult = RunResult(
        ok = r.ok,
        stdout = r.stdout,
        stderr = "",
        errorType = r.errorType,
        errorMessage = r.errorMessage,
        traceback = null,
        durationMs = (System.nanoTime() - startedAt) / 1_000_000,
        variables = r.variables.map { VarInfo(it.first, it.second, it.third) },
        passed = null
    )

    COMFORT("comfort", "舒适", 1.15f),
    LARGE("large", "大字", 1.3f);

        fun fromId(id: String?): FontScale = entries.firstOrNull { it.id == id } ?: DEFAULT
    }

// 不然中文标题在窄屏上会乱换行。
fun typographyFor(scale: FontScale): Typography {
    if (scale == FontScale.NORMAL) return AppTypography
    val f = scale.factor
    fun TextStyle.scaled(): TextStyle = copy(
        fontSize = (fontSize.value * f).sp,
        lineHeight = (lineHeight.value * f).sp
    )
    // 只放大正文，标题和标签保持原样。
    return AppTypography.copy(
        bodyLarge = AppTypography.bodyLarge.scaled(),
        bodyMedium = AppTypography.bodyMedium.scaled(),
        bodySmall = AppTypography.bodySmall.scaled()
    )
}

    const val DEFAULT_USER_AGENT = "CodeLearning-Android"
    private const val DEFAULT_TIMEOUT_MS = 6000
    private const val DEFAULT_MAX_BYTES = 5 * 1024 * 1024

    fun getText(
        url: String,
        timeoutMs: Int = DEFAULT_TIMEOUT_MS,
        maxBytes: Int = DEFAULT_MAX_BYTES,
        accept: String = "application/json",
        userAgent: String = DEFAULT_USER_AGENT
    ): String = String(getBytes(url, timeoutMs, maxBytes, accept, userAgent), StandardCharsets.UTF_8)

    fun getBytes(
        url: String,
        timeoutMs: Int = DEFAULT_TIMEOUT_MS,
        maxBytes: Int = DEFAULT_MAX_BYTES,
        accept: String = "application/octet-stream",
        userAgent: String = DEFAULT_USER_AGENT
    ): ByteArray {
        val conn = (URL(url).openConnection() as HttpURLConnection).apply {
            connectTimeout = timeoutMs
            readTimeout = timeoutMs
            requestMethod = "GET"
            setRequestProperty("Accept", accept)
            setRequestProperty("User-Agent", userAgent)

CodeLearning（码上）多语言编程学习软件 V1.0 第 40 页 [app/src/main/java/com/pyneon/academy/utils/HttpText.kt 自第 39 行起]
        }
        try {
            if (conn.responseCode !in 200..299) {
                throw IOException("HTTP ${conn.responseCode} ($url)")
            }
            return conn.inputStream.use { readCapped(it, maxBytes, url) }
        } finally {
            conn.disconnect()
        }
    }

    private fun readCapped(stream: InputStream, maxBytes: Int, url: String): ByteArray {
        val buf = ByteArray(64 * 1024)
        var total = 0
        val out = ByteArrayOutputStream(256 * 1024)
        while (true) {
            val n = stream.read(buf)
            if (n < 0) break
            total += n
            if (total > maxBytes) {
                throw IOException("HTTP 响应超过 ${maxBytes / (1024 * 1024)}MB，已断开 ($url)")
            }
            out.write(buf, 0, n)
        }
        return out.toByteArray()
    }
}

class _Watchdog:

    def __call__(self, frame, event, arg):

def _error_payload(etype, evalue, tb):
    trimmed_tb = tb.tb_next if tb is not None else None

"""设备端 SQLite 内存库的 SQL 执行和判题。

SQLite 的 execute 是同步阻塞的，死循环 CTE（比如漏了终止条件的 WITH RECURSIVE）
会把线程占死，界面一直转。这里用 set_progress_handler 做指令级看门狗：
每 1000 条虚拟机指令看一眼时间，超了返回非 0，SQLite 自己中断并抛错。
"""
import json
import sqlite3
import time
import traceback

DEFAULT_TIMEOUT = 8.0
MAX_OUTPUT = 1_048_576
MAX_ROWS = 200
# 每多少条 SQLite 虚拟机指令检查一次时钟。数值越小越及时、开销越大；
# 1000 在低端机上对常见查询的开销可忽略，中断粒度约毫秒级。
_PROGRESS_STEPS = 1000

_DENIED_ACTIONS = {
    getattr(sqlite3, "SQLITE_ATTACH", 24),
    getattr(sqlite3, "SQLITE_DETACH", 25),
}


def _authorizer(action, arg1, arg2, dbname, source):
    if action in _DENIED_ACTIONS:

    return sqlite3.SQLITE_OK


def _split_statements(sql):
    """按分号切分语句，忽略字符串与 -- 行注释中的分号。"""
    statements = []
    buf = []
    quote = None
    i = 0
    while i < len(sql):
        ch = sql[i]
        if quote:
            buf.append(ch)
            if ch == quote:
                if i + 1 < len(sql) and sql[i + 1] == quote:
                    buf.append(sql[i + 1])
                    i += 2
                    continue
                quote = None
            i += 1
            continue
        if ch in ("'", '"'):
            quote = ch
            buf.append(ch)
            i += 1
            continue
        if ch == "-" and i + 1 < len(sql) and sql[i + 1] == "-":
            while i < len(sql) and sql[i] != "\n":
                i += 1
            continue
        if ch == ";":
            text = "".join(buf).strip()
            if text:
                statements.append(text)
            buf = []
            i += 1
            continue
        buf.append(ch)
        i += 1
    tail = "".join(buf).strip()
    if tail:
        statements.append(tail)
    return statements

def _execute(sql, timeout=None):
    started = time.perf_counter()
    budget = DEFAULT_TIMEOUT if timeout is None else float(timeout)
    deadline = started + budget
    timed_out = {"hit": False}

    def _watchdog():
        """SQLite 每执行 _PROGRESS_STEPS 条指令回调一次；返回非 0 中断语句。"""
        if time.perf_counter() > deadline:
            timed_out["hit"] = True
            return 1
        return 0

    # isolation_level=None 关闭 Python 的隐式事务管理：否则 sqlite3 会在每条语句前
    # 自动 BEGIN，用户写的 BEGIN / COMMIT / ROLLBACK 会直接报
    # 「cannot start a transaction within a transaction」。
    # 关闭后行为与 sqlite3 命令行一致，事务控制权完全交给 SQL 本身。
    conn = sqlite3.connect(":memory:", isolation_level=None)
    conn.set_authorizer(_authorizer)
    conn.set_progress_handler(_watchdog, _PROGRESS_STEPS)
    out = []
    error = None
    last = None
    try:
        for statement in _split_statements(sql):
            cursor = conn.execute(statement)
            if cursor.description:
                columns, rows, text = _format_table(cursor)
                last = {
                    "columns": columns,
                    "rows": [[None if v is None else v for v in row] for row in rows],
                }
                out.append(text)
            else:
                out.append("OK (%d rows affected)" % cursor.rowcount)
        conn.commit()
    except BaseException:  # 任何异常都要转成结构化结果，别让桥接层崩。
        try:
            conn.rollback()  # 未提交的事务在报错时必须撤销，避免半成品状态
        except Exception:
            pass
        etype, evalue, tb = __import__("sys").exc_info()
        trimmed = tb.tb_next if tb is not None else None
        if timed_out["hit"]:
            # SQLite 中断时抛的是 OperationalError('interrupted')，对外统一报超时，

CodeLearning（码上）多语言编程学习软件 V1.0 第 58 页 [app/src/main/python/sql_runner.py 自第 127 行起]
            # 不然用户只看到一句看不懂的 interrupted。
            error = {
                "type": "Timeout",
                "message": "执行超过 %.1f 秒上限，已中断（疑似死循环或超大查询）" % budget,
                "traceback": "",
            }
        else:
            error = {
                "type": getattr(etype, "__name__", "Error"),
                "message": str(evalue),
                "traceback": "".join(traceback.format_exception(etype, evalue, trimmed)).strip(),
            }
    finally:
        conn.close()
    text = "\n".join(out)
    if len(text) > MAX_OUTPUT:
        text = "[输出过长已截断]\n" + text[-MAX_OUTPUT:]
    return {
        "ok": error is None,
        "stdout": text,
        "stderr": "",
        "error": error,
        "duration_ms": int((time.perf_counter() - started) * 1000),
        "last": last,
    }
