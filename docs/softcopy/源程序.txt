class MainActivity : ComponentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge()
        setContent {
            var currentTheme by remember { mutableStateOf(ThemePreference.getCurrentTheme(this)) }
            var visualStyle by remember { mutableStateOf(ThemePreference.getVisualStyle(this)) }
            var fontScale by remember { mutableStateOf(ThemePreference.getFontScale(this)) }
            // 主题 / 风格偏好变更即时生效（SharedPreferences 监听，无需重启应用）
            DisposableEffect(Unit) {
                val listener =
                    android.content.SharedPreferences.OnSharedPreferenceChangeListener { _, key ->
                        when (key) {
                            ThemePreference.KEY_THEME_ID ->
                                currentTheme = ThemePreference.getCurrentTheme(this@MainActivity)
                            ThemePreference.KEY_VISUAL_STYLE ->
                                visualStyle = ThemePreference.getVisualStyle(this@MainActivity)
                            ThemePreference.KEY_FONT_SCALE ->
                                fontScale = ThemePreference.getFontScale(this@MainActivity)
                        }
                    }
                ThemePreference.registerListener(this@MainActivity, listener)
                onDispose { ThemePreference.unregisterListener(this@MainActivity, listener) }
            }
            PyNeonTheme(appTheme = currentTheme, visualStyle = visualStyle, fontScale = fontScale) {
                AppRoot()
            }
        }
    }
}

        "if ", "elif ", "else", "for ", "while ", "def ", "class ",
        "try", "except", "finally", "with ", "match ", "case "
    )

    fun hints(code: String, language: String): List<String> {
        if (code.isBlank()) return emptyList()
        val scan = scan(code, language)
        val hints = scan.hints.toMutableList()

        val lines = scan.cleaned.lines()
        if (language == "python") {
            lines.forEachIndexed { index, line ->
                val trimmed = line.trim()
                val looksLikeBlock = PYTHON_BLOCK_WORDS.any { trimmed.startsWith(it) } ||
                    trimmed.endsWith("else") || trimmed.endsWith("try") || trimmed.endsWith("finally")
                if (looksLikeBlock && !trimmed.endsWith(":") && !trimmed.contains("#")) {

    private fun search(
        question: String,
        sources: List<TutorSource>,
        currentLessonId: String
    ): List<TutorSource> {
        val terms = searchTerms(question)
        if (terms.isEmpty()) return emptyList()
        return sources.asSequence()
            .map { source ->
                var score = 0
                if (source.title.contains(question, ignoreCase = true)) score += 6
                if (source.text.contains(question, ignoreCase = true)) score += 8
                for (term in terms) {
                    score += source.title.countOccurrences(term) * 4
                    score += source.text.countOccurrences(term)
                }
                if (source.lessonId == currentLessonId) score += 2
                source to score
            }
            .filter { it.second > 0 }
            .sortedByDescending { it.second }
            .map { it.first }
            .toList()
    }

    private fun searchTerms(question: String): List<String> {
        val cleaned = question.lowercase()
            .replace(Regex("[\\s，。？！、；：,.?!;:（）()\\[\\]【】]+"), " ")
        val latin = Regex("[a-z0-9_]{2,}").findAll(cleaned).map { it.value }.toList()
        val chinese = Regex("[\\u4e00-\\u9fa5]{2,}").findAll(cleaned).map { it.value }.toList()
        val grams = chinese.flatMap { word ->
            if (word.length <= 4) {
                listOf(word)
            } else {
                // 长问题只取前八个二字词会漏掉句尾的关键词，头尾各取四个。
                val pairs = word.windowed(2)
                (pairs.take(4) + pairs.takeLast(4)).distinct()
            }
        }
        return (latin + grams)
            .filterNot { it in STOP_TERMS }
            .distinct()
            .take(16)
    }

        fun fromId(id: String?): FontScale = entries.firstOrNull { it.id == id } ?: DEFAULT
    }

// 按档位生成排版，只放大三档正文。标题和标签不动，

fun typographyFor(scale: FontScale): Typography {
    if (scale == FontScale.NORMAL) return AppTypography
    val f = scale.factor
    fun TextStyle.scaled(): TextStyle = copy(
        fontSize = (fontSize.value * f).sp,
        lineHeight = (lineHeight.value * f).sp
    )
    // 只放大正文，标题和标签保持原样。
    return AppTypography.copy(
        bodyLarge = AppTypography.bodyLarge.scaled(),
        bodyMedium = AppTypography.bodyMedium.scaled(),
        bodySmall = AppTypography.bodySmall.scaled()
    )
}

            setRequestProperty("User-Agent", userAgent)
        }
        try {
            if (conn.responseCode !in 200..299) {
                throw IOException("HTTP ${conn.responseCode} ($url)")
            }
            return conn.inputStream.use { readCapped(it, maxBytes, url) }
        } finally {
            conn.disconnect()
        }
    }

    private fun readCapped(stream: InputStream, maxBytes: Int, url: String): ByteArray {
        val buf = ByteArray(64 * 1024)
        var total = 0
        val out = ByteArrayOutputStream(256 * 1024)
        while (true) {
            val n = stream.read(buf)
            if (n < 0) break
            total += n
            if (total > maxBytes) {
                throw IOException("HTTP 响应超过 ${maxBytes / (1024 * 1024)}MB，已断开 ($url)")
            }
            out.write(buf, 0, n)
        }
        return out.toByteArray()
    }
}

@contextmanager
def sandbox_guard():
    """执行用户代码期间打开操作级沙箱；可重入，按线程隔离。"""
    tid = threading.get_ident()

class _BoundedBuffer(io.StringIO):
    """带上限的输出缓冲。只在结束后截断挡不住内存，
    死循环 print 在超时前就能吃几百 MB。这里边写边量，超过上限直接抛异常。"""

    HARD_LIMIT = MAX_OUTPUT * 4

        self._written = 0

    def write(self, s):
        self._written += len(s)
        if self._written > self.HARD_LIMIT:
            raise OutputTooLong()
        return super().write(s)

"""设备端 SQLite 内存库的 SQL 执行和判题。

SQLite 的 execute 是同步阻塞的，死循环 CTE（比如漏了终止条件的 WITH RECURSIVE）
会把线程占死，界面一直转。这里用 set_progress_handler 做指令级看门狗：
每 1000 条虚拟机指令看一眼时间，超了返回非 0，SQLite 自己中断并抛错。
"""
import json
import sqlite3
import time
import traceback

DEFAULT_TIMEOUT = 8.0
MAX_OUTPUT = 1_048_576
MAX_ROWS = 200
# 每多少条 SQLite 虚拟机指令检查一次时钟。数值越小越及时、开销越大；
# 1000 在低端机上对常见查询的开销可忽略，中断粒度约毫秒级。
_PROGRESS_STEPS = 1000

_DENIED_ACTIONS = {
    getattr(sqlite3, "SQLITE_ATTACH", 24),
    getattr(sqlite3, "SQLITE_DETACH", 25),
}


def _authorizer(action, arg1, arg2, dbname, source):
    if action in _DENIED_ACTIONS:
        return sqlite3.SQLITE_DENY
    return sqlite3.SQLITE_OK


def _split_statements(sql):
    """按分号切分语句，忽略字符串与 -- 行注释中的分号。"""
    statements = []
    buf = []
    quote = None

    i = 0
    while i < len(sql):

        if quote:
            buf.append(ch)

                    buf.append(sql[i + 1])
                    i += 2

            i += 1

def _format_table(cursor):
    columns = [d[0] for d in cursor.description or []]
    rows = cursor.fetchmany(MAX_ROWS)
    lines = ["\t".join(columns)] if columns else []
    for row in rows:
        lines.append("\t".join("NULL" if v is None else str(v) for v in row))
    return columns, rows, "\n".join(lines)
